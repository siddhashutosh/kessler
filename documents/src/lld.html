3. Logic Layer (app/logic) — algorithms

3.1 cdm_parser.py

@dataclass CdmObject: designator, name, object_type, r_rtn: vec3|None,
                      v_rtn: vec3|None, cov_rtn: 3x3|None      # metres, m/s, m²
@dataclass Cdm: cdm_id, created, tca: datetime, miss_distance_m, relative_speed_ms,
                sat1: CdmObject, sat2: CdmObject, pc_reported: float|None,
                emergency_reportable: bool, source: str, raw: dict

parse_cdm_kvn(text: str) -> Cdm          # CCSDS 508.0-B-1 KVN: header → relative
                                         # metadata → OBJECT1/OBJECT2 blocks (state+cov)
parse_cdm_public_row(row: dict) -> Cdm   # Space-Track cdm_public JSON row
                                         # (CDM_ID, CREATED, TCA, MIN_RNG, PC,
                                         #  SAT_1_ID, SAT_1_NAME, SAT1_OBJECT_TYPE, …)

Validation: TCA parseable and TZ-aware, both designators present, numeric fields finite, covariance symmetric-PSD when supplied (symmetrised; non-PSD → flagged, dropped to max-Pc path). All violations accumulated then raised in one CdmParseError(detail=[...]) (FR-CDM-3).

3.2 propagation.py

load_satrec(omm: dict) -> Satrec                     # sgp4.exporter-compatible OMM load
state_teme(sat, t: datetime) -> (r_km, v_kms)        # raises PropagationError on sgp4 code≠0
sample_range(satA, satB, t0, t1, step_s) -> ndarray  # vectorised |rA-rB| over window
refine_tca(satA, satB, t_guess, half_window_s=90) -> (tca, miss_m, rel_speed)
    # golden-section minimisation of range(t), tol 0.25 s → then quadratic fit for sub-s TCA

3.3 collision_probability.py — the mathematical core (FR-PC)

encounter_plane_projection(r_rel, v_rel, cov1, cov2) -> (mu_2d, C_2d)
    # 1. combined covariance C = cov1 + cov2 (independent errors)
    # 2. basis: ez = v_rel/|v_rel|; ex = r_rel − (r_rel·ez)ez, normalised; ey = ez × ex
    # 3. project miss vector and C into (ex, ey) plane → mu (2,), C_2d (2×2)

pc_foster(mu, C, hbr_m, n_r=60, n_theta=90) -> float
    # numerical polar integration of bivariate Gaussian over disc radius=hbr
    # Pc = ∬ N(x; mu, C) dA, midpoint rule in (r, θ); relative err < 1e-3 for n defaults

pc_chan(mu, C, hbr_m, terms=20) -> float
    # Chan (2008) equivalent-area series: transform to isotropic space,
    # u = hbr²/(σx·σy·…), v = miss²…; Pc = e^{−v/2} Σ (v/2)^m/m! · [1 − e^{−u/2} Σ…]

pc_max(miss_m, hbr_m) -> float
    # covariance-free conservative bound (Alfano style):
    # maximise isotropic-σ Pc over σ → σ* = miss/√2 → Pc_max = (hbr²/miss²)·e^{−1}
    # (clipped to 1; documented as triage-grade upper bound)

compute_pc(cdm, hbr_m) -> PcResult(pc, method, cross_check, divergence_flag, pc_type)
    # dispatch: covariance → foster + chan cross-check (|Δ|/max > 5% ⇒ flag)
    #           no covariance → pc_max, pc_type="max"

3.4 screening.py

screen(asset_omm, catalog: list[omm], window_h, threshold_km=10) -> list[CloseApproach]
  Step 1  altitude sieve: skip object if [perigee,apogee] bands are > threshold apart
  Step 2  coarse scan: shared time grid (60 s), vectorised ranges; local minima < 5×threshold
  Step 3  refine each candidate with refine_tca(); keep miss ≤ threshold
  Step 4  annotate RTN components of miss vector, rel speed, pc_max

3.5 risk_engine.py

classify(pc, pc_type, miss_m, cov_available) -> RiskClass   # FR-RSK-1 thresholds
urgency(risk, tca, created) -> float 0..100
    # base by class (CRITICAL 70) + up to 20 for time-to-TCA < 24 h + up to 10 for stale data
recommend(risk, tca, cov_available) -> str                  # deterministic action text

6. Service Layer Detail

6.1 spacetrack_client.py — rate-limit governor

class SpaceTrackClient:
    MIN_INTERVAL_S = 3.0          # ≥ 3 s between calls ⇒ < 20/min (CON-1 margin)
    HOURLY_BUDGET  = 250          # < 300/hr with margin
    async login()                 # POST /ajaxauth/login, session cookie
    async fetch_cdm_public(limit=200) -> list[dict]
        # /basicspacedata/query/class/cdm_public/orderby/TCA asc/limit/{n}/format/json
    async fetch_gp(norad_ids)     # only if cache-miss AND ttl expired (guarded by cache_service)
    # every call: acquire throttle lock → check budget → httpx call (timeout 30 s,
    # 2 retries exp backoff) → on 401 re-login once → errors → DataSourceError/RateLimitError

6.2 conjunction_service.py — pipeline orchestration (sequence)

refresh_pipeline():                                  # startup + scheduled
  1 pipeline.start("catalog_sync") → gp = cache.get_or_fetch("gp:active", loader)
  2 pipeline.start("cdm_ingest")   → raw = cache.get_or_fetch("cdm:latest", loader|demo)
  3 pipeline.start("parse")        → cdms = [parse_cdm_public_row(r)] ; quarantine failures
  4 pipeline.start("pc_engine")    → for each: compute_pc(cdm, settings.default_hbr_m)
  5 pipeline.start("risk")         → classify + urgency + recommend (+ insight lazily)
  6 pipeline.start("publish")      → upsert events table; snapshot generated_at
  each stage: try/except → pipeline.finish(status, items, error); stage failure uses
  last-known-good data and marks agent "degraded" — pipeline never aborts the app.

6.3 insight_service.py

If anthropic_api_key set: Claude Messages API call (short, temperature 0.2, max_tokens 400) with event JSON context → analyst briefing; 10 s timeout; on any exception log WARN and return template briefing (source:"template"). Never raises to caller (FR-RSK-4).
